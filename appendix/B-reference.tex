\paragraph{Useful identities.}
\begin{align*}
  &\sigma_j\sigma_k=\delta_{jk}\Id+\ii\sum_l\varepsilon_{jkl}\sigma_l,\qquad
  [\sigma_j,\sigma_k]=2\ii\sum_l\varepsilon_{jkl}\sigma_l,\qquad\{\sigma_j,\sigma_k\}=2\delta_{jk}\Id,\\
  &H=\tfrac{1}{\sqrt2}(X+Z),\quad H^2=\Id,\quad HXH=Z,\quad HZH=X,\quad HYH=-Y,\\
  &e^{-\ii\theta\,\vec n\cdot\vec\sigma/2}=\cos\tfrac\theta2\,\Id-\ii\sin\tfrac\theta2\,\vec n\cdot\vec\sigma,\\
  &\ket\psi\!\bra\psi=\tfrac12(\Id+\vec n\cdot\vec\sigma),\qquad \abs{\braket\varphi\psi}^2=\tfrac12(1+\vec m\cdot\vec n).
\end{align*}

\paragraph{The postulates in one line each.}
(1)~A state is a unit vector $\ket\psi\in\Hil$.
(2)~Closed systems evolve by $\ket\psi\mapsto U\ket\psi$, with $U=e^{-\ii Ht/\hbar}$.
(3)~A measurement $\{M_m\}$ with $\sum_mM_m^\dagger M_m=\Id$ gives $m$ with probability $\bra\psi M_m^\dagger M_m\ket\psi$ and leaves $M_m\ket\psi/\sqrt{p(m)}$.
(4)~The state space of a composite system is $\Hil_1\tp\Hil_2$.
