%======================================================================
%  boxes.tex -- the visual language of the book.
%
%  Deliberately quiet: one accent colour (navy), greys, and a single
%  muted red reserved for pitfalls. What is what:
%
%    Definition .......... navy bar, pale blue fill     numbered
%    Theorem, Lemma,
%    Proposition,
%    Corollary ........... navy bar, darker fill        numbered
%    Proof ............... plain text, "Proof." and a filled square
%    Example ............. grey bar, grey fill          numbered
%    Solution ............ same grey bar, no fill (it continues the example)
%    Remark .............. plain text, "Remark."
%    Pitfall ............. muted red box                 (a classic mistake)
%    Cross-reference,
%    Where this sits ..... thin navy rule, small text
%    Learning objectives . navy rules above and below
%    Exercises ........... gold stars, listed at the end of each section
%======================================================================

% ---------- Colour palette ----------
\definecolor{navy}{RGB}{0,45,100}
\definecolor{ink}{RGB}{70,70,70}         % greys
\definecolor{warncol}{RGB}{150,45,45}    % pitfalls only
\definecolor{starcol}{RGB}{190,140,0}    % exercise stars

% ---------- Keeping short blocks on one page ----------
%  \keepspace{<length>} starts a new page unless <length> of space is left on the
%  current one.  It measures the space left (\pagegoal - \pagetotal), like the
%  \Needspace command of the needspace package.  (The glue trick of \needspace is
%  not used: its stretchable glue makes the break before the block look cheaper
%  to TeX than any later one, and pages end with large holes.)  Right after a
%  heading it does nothing, so that a heading is never separated from what follows.
%  The skip that preceded it is put back last, so that a following \addvspace
%  (used by proofs and by the boxes) still sees it as \lastskip.
\newcommand{\keepspace}[1]{%
  \par
  \if@nobreak\else
    \@tempskipb\lastskip
    \vskip-\@tempskipb
    \penalty\z@          % lets the page builder catch up, so \pagetotal is current
    \begingroup
      \setlength\dimen@{#1}%
      \advance\dimen@\@tempskipb
      \dimen@ii\pagegoal
      \advance\dimen@ii-\pagetotal
      \ifdim\dimen@>\dimen@ii
        \ifdim\dimen@ii>\z@ \vfil\fi
        \break
      \else
        \vskip\@tempskipb
      \fi
    \endgroup
  \fi
}

% ---------- Shared look for the framed environments ----------
\tcbset{
  bookbox/.style={
    enhanced, breakable, parbox=false, frame hidden, boxrule=0pt, sharp corners,
    left=10pt, right=8pt, top=5pt, bottom=5pt,
    before skip=0.8em, after skip=0.8em,
  },
}

% Inline label: "Definition 1.3 (Optional name)."
\newcommand{\boxlabel}[3]{\textbf{\sffamily\color{#1}#2~\thechapter.\arabic{result}}%
  \ifstrempty{#3}{}{\ {\sffamily(#3)}}\textbf{\sffamily .}\ }

% ---------- Numbered statement boxes ----------
\newcommand{\newstatementbox}[4]{% env, Label, bar colour, fill
  \newtcolorbox[use counter=result]{#1}[1][]{%
    bookbox, unbreakable, colback=#4, borderline west={2.5pt}{0pt}{#3},
    before upper={\xdef\@currentlabel{\thechapter.\arabic{result}}%
                  \boxlabel{#3}{#2}{##1}},
  }%
}
\newstatementbox{definition}{Definition}{navy}{navy!5}
\newstatementbox{theorem}{Theorem}{navy}{navy!10}
\newstatementbox{lemma}{Lemma}{navy}{navy!10}
\newstatementbox{proposition}{Proposition}{navy}{navy!10}
\newstatementbox{corollary}{Corollary}{navy}{navy!10}

% ---------- Postulates: their own counter, a heavier navy block ----------
\newcounter{postulate}
\newtcolorbox{postulate}[1][]{%
  bookbox, unbreakable, colback=navy!14, borderline west={4pt}{0pt}{navy},
  before upper={\refstepcounter{postulate}%
                \textbf{\sffamily\color{navy}Postulate~\arabic{postulate}%
                \ifstrempty{#1}{}{\ {\normalfont\sffamily(#1)}}.}\ },
}

% ---------- Example (statement) and Solution: one continuous grey bar ----------
\newtcolorbox[use counter=result]{examplebox}[1][]{%
  bookbox, unbreakable, colback=ink!7, borderline west={2.5pt}{0pt}{ink},
  after skip=0pt, bottom=2pt,
  before upper={\xdef\@currentlabel{\thechapter.\arabic{result}}%
                \boxlabel{ink}{Example}{#1}},
}
% The statement must not be left alone at the foot of a page, away from its
% solution.  So the statement is first set in a scratch box (never printed: its
% labels are not written and its counter step is undone) to measure its height,
% and \keepspace asks for that height plus five lines ("Solution." and the four
% lines that a solution needs before it may break).
\newsavebox{\ex@measure}
\NewDocumentEnvironment{example}{O{} +b}{%
  \edef\ex@result{\the\value{result}}%
  \sbox{\ex@measure}{\parbox{\linewidth}{\begin{examplebox}[#1]#2\end{examplebox}}}%
  \setcounter{result}{\ex@result}%
  \keepspace{\dimexpr\ht\ex@measure+\dp\ex@measure+5\baselineskip\relax}%
  \begin{examplebox}[#1]#2\end{examplebox}%
}{}
\newtcolorbox{solution}{%
  bookbox, colback=white, borderline west={2.5pt}{0pt}{ink},
  before skip=0pt, top=2pt, lines before break=4,
  before upper={\@endparpenalty=\@M\relax\textit{Solution.}\ },
  after upper={\qedmark{\square}},
}

% ---------- Pitfall: the only warm colour ----------
\newtcolorbox{pitfall}{%
  bookbox, unbreakable, frame hidden, colback=warncol!6, borderline west={2.5pt}{0pt}{warncol},
  before upper={\textbf{\sffamily\color{warncol}Pitfall.}\ },
}

% ---------- Quiet notes: cross-reference and place in the course ----------
\newcommand{\newnote}[2]{%
  \newtcolorbox{#1}{%
    bookbox, unbreakable, colback=white, borderline west={1pt}{0pt}{navy!60},
    fontupper=\small, top=2pt, bottom=2pt,
    before upper={\textbf{\sffamily\color{navy}#2.}\ },
  }%
}
\newnote{xref}{Cross-reference}
\newnote{whereitsits}{Where this sits}
\newnote{lookahead}{Looking ahead}
\newnote{recall}{Recall}

% ---------- Step-by-step lists for proofs, solutions and explanations ----------
% \begin{steps} \item ... \end{steps}   (a short, ordered chain of reasoning)
% \begin{steps*} \item ... \end{steps*} (the same, numbered)
\newlist{steps}{itemize}{2}
\setlist[steps,1]{label=\textcolor{navy}{\small$\blacktriangleright$}, leftmargin=1.5em, itemsep=0.35em, topsep=0.3em, parsep=0pt}
\setlist[steps,2]{label=\textcolor{ink}{\scriptsize$\bullet$}, leftmargin=1.3em, itemsep=0.2em, topsep=0.2em, parsep=0pt}
\newlist{stepsn}{enumerate}{2}
\setlist[stepsn,1]{label=\textcolor{navy}{\small\bfseries\arabic*.}, leftmargin=1.7em, itemsep=0.35em, topsep=0.3em, parsep=0pt}
\setlist[stepsn,2]{label=\textcolor{ink}{(\alph*)}, leftmargin=1.5em, itemsep=0.2em, topsep=0.2em, parsep=0pt}

% ---------- Remark and proof: plain text ----------
\newcommand{\qedmark}[1]{\leavevmode\unskip\penalty9999 \hbox{}\nobreak\hfill\quad$#1$}

\renewenvironment{proof}[1][Proof]{\keepspace{3\baselineskip}\addvspace{0.4em}\@endparpenalty=\@M\relax\noindent\textit{#1.}\ }%
  {\qedmark{\blacksquare}\par\addvspace{0.6em}}
\providecommand{\qedhere}{}\renewcommand{\qedhere}{}   % the end mark is added automatically
\newenvironment{remark}{\par\addvspace{0.3em}\noindent\textit{Remark.}\ }%
  {\par\addvspace{0.3em}}

% ---------- Exercises (no solutions), graded by stars ----------
\newcommand{\stars}[1]{\textcolor{starcol}{%
  \ifcase#1\or$\star$\or$\star\star$\or$\star\star\star$\fi}}

\newenvironment{exerciseset}{%
  \keepspace{6\baselineskip}\vspace{1.2em}\noindent
  {\sffamily\bfseries\large Exercises}\hfill
  {\footnotesize\sffamily\color{ink} $\star$ routine \quad $\star\star$ needs an idea \quad
   $\star\star\star$ challenging}\par\nobreak
  \vspace{0.1em}{\color{ink}\hrule height 0.6pt}\vspace{0.3em}
}{\par\vspace{0.3em}{\color{ink}\hrule height 0.3pt}\vspace{0.8em}}
