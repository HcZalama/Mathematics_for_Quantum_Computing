%======================================================================
%  theorems.tex -- counters and end-of-proof markers.
%  The boxes themselves (definition, theorem, example, ...) are in boxes.tex.
%
%  All numbered boxes share ONE counter, 'result', numbered within the
%  chapter: Definition 1.1, Example 1.2, Theorem 1.3, ...  One sequence per
%  chapter means "Theorem 1.7" is always findable by counting up.
%======================================================================

\newcounter{result}[chapter]
\renewcommand{\theresult}{\thechapter.\arabic{result}}

% Exercises restart in every section: Exercise 1.2.3 = chapter 1, section 2, third.
\newcounter{exercise}[section]
\renewcommand{\theexercise}{\thesection.\arabic{exercise}}

% End-of-statement markers, for the rare place where a proof is not in a box:
% hollow = quoted without proof, filled = end of an argument.
\newcommand{\qedopen}{\par\nobreak\hfill$\square$}
\newcommand{\qedfilled}{\par\nobreak\hfill$\blacksquare$}
